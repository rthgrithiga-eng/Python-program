# Python-program
Python using list, tuple 
# List (mutable - can change)
numbers = [10, 20, 30, 40]

# Tuple (immutable - cannot change)
values = (1, 2, 3, 4)

# Access elements
print("List element:", numbers[1])
print("Tuple element:", values[2])

# Modify list
numbers.append(50)

print("Updated List:", numbers)
print("Tuple:", values)
